[step]
# 进度页的步骤名。ROTATE_TO_BAY 的 %s 是方位名，LIFT_TO_LAYER 的 %d 是层号。
NONE = 准备中
FETCH_TRAY = 调取空托盘到出入口
PULL_FROM_DOCK = 从出入口拉回托盘
ROTATE_TO_BAY = 旋转至%s位
LIFT_TO_LAYER = 升降至 %dF
ENGAGE = 升至啮合高度
PUSH_INTO_BAY = 送入车位
FORK_RETRACT = 拨叉收回
PRESET_TRAY = 预置空托盘到出入口
RETURN_DOCK_TRAY = 送回出入口空托盘
PULL_FROM_BAY = 取出载车托盘
ROTATE_TO_DOCK = 旋转至出入口
PUSH_TO_DOCK = 送到出入口
HOMING = 回零中

[fmt]
JOB_STORE = 存车中 · %dF %s
JOB_GET = 取车中 · %dF %s

BAY = %dF · %s
FAULT_AT = %dF %s位 · 第 %d 步\r

[fault]
# 故障码 = 标题 | 处理提示
1 = 位置未确认 | 上电或急停后位置未知\r确认下面两项后回零
10 = 升降超时 | 升降轴没有按时到位\r请检查升降机构和下限位开关
11 = 旋转超时 | 旋转轴没有按时到位\r请检查旋转机构和回零检测
12 = 拨叉伸出超时 | 请检查托盘是否卡在交接缝上
13 = 拨叉收回超时 | 请检查拨叉和收回位开关
20 = 车位状态不符 | 霍尔检测与车位记录不一致\r请核对托盘实际位置
21 = 转运台托盘异常 | 转运台霍尔状态与动作不符
30 = 限位异常 | 运动中触发了限位开关
40 = 层检测板离线 | 层板没有应答\r请检查层板接线和供电
41 = 车位未标定 | 请进入维护页完成标定
0 = 未知故障 | 请记录故障码并联系维护人员
